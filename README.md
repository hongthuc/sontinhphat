    /* Render toàn bộ 7 Danh mục sản phẩm ra Trang Chủ theo chuẩn Flatsome */
    function renderAllProductSections(filterKeyword = '') {
      const container = document.getElementById('productSectionsContainer');
      const keyword = filterKeyword.trim().toLowerCase();

      let html = '';

      PRODUCT_SECTIONS.forEach((section) => {
        const filteredProducts = section.products.filter((p) => {
          if (!keyword) return true;
          return (
            p.name.toLowerCase().includes(keyword) ||
            p.subcat.toLowerCase().includes(keyword) ||
            section.title.toLowerCase().includes(keyword)
          );
        });

        if (filteredProducts.length === 0) return;

        html += `
          <section id="${section.anchor}" class="flatsome-container">
            <div class="bg-white rounded-lg border border-slate-200 p-3.5 md:p-5 shadow-sm">
              <!-- Thanh tiêu đề danh mục chuẩn Flatsome (chữ nhỏ gọn, viết liền trên 1 dòng) -->
              <div class="flex flex-wrap items-center justify-between gap-2 border-b-2 border-stpBlue pb-2.5 mb-3.5">
                <div class="flex items-center gap-2">
                  <span class="w-2 h-5 bg-stpOrange rounded-sm inline-block"></span>
                  <h2 class="text-sm sm:text-base md:text-lg font-extrabold text-stpBlue uppercase whitespace-nowrap">
                    ${section.title}
                  </h2>
                </div>

                <!-- Các nút lọc danh mục con nằm gọn trên 1 hàng -->
                <div class="flex items-center gap-1.5 overflow-x-auto no-scrollbar py-0.5">
                  <button
                    type="button"
                    onclick="filterSectionProducts('${section.id}', 'ALL', this)"
                    class="subcat-pill active"
                  >
                    Tất cả
                  </button>
                  ${section.subcategories
                    .map(
                      (sub) => `
                    <button
                      type="button"
                      onclick="filterSectionProducts('${section.id}', '${sub}', this)"
                      class="subcat-pill"
                    >
                      ${sub}
                    </button>
                  `
                    )
                    .join('')}
                </div>
              </div>

              <!-- Lưới sản phẩm: Điện thoại chia 2 cột (grid-cols-2), Máy tính chia 5 cột (lg:grid-cols-5) -->
              <div id="grid-${section.id}" class="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-5 gap-2.5 md:gap-3.5">
                ${filteredProducts.map((prod) => buildProductCardHtml(prod, section.title)).join('')}
              </div>
            </div>
          </section>
        `;
      });

      if (!html) {
        html = `
          <div class="flatsome-container">
            <div class="bg-white rounded-lg border border-slate-200 p-8 text-center space-y-3">
              <div class="text-sm font-bold text-slate-700">Không tìm thấy sản phẩm phù hợp với từ khóa "${filterKeyword}"</div>

                Xem lại tất cả sản phẩm
              </button>
            </div>
          </div>
        `;
      }

      container.innerHTML = html;
    }

    /* Lọc sản phẩm theo danh mục con ngay trong từng Section */
    function filterSectionProducts(sectionId, subcat, btnEl) {
      const section = PRODUCT_SECTIONS.find((s) => s.id === sectionId);
      if (!section) return;

      const gridEl = document.getElementById(`grid-${sectionId}`);
      if (!gridEl) return;

      if (btnEl && btnEl.parentElement) {
        Array.from(btnEl.parentElement.children).forEach((b) => b.classList.remove('active'));
        btnEl.classList.add('active');
      }

      const list =
        subcat === 'ALL'
          ? section.products
          : section.products.filter((p) => p.subcat.toLowerCase().includes(subcat.toLowerCase()));

      gridEl.innerHTML = list.map((prod) => buildProductCardHtml(prod, section.title)).join('');
    }

    /* Điều khiển 2 Banner Tự Trượt Bằng Nhau */
    let currentBannerIndex = 0;
    const totalBanners = 2;
    let bannerInterval = null;

    function showBannerSlide(index) {
      currentBannerIndex = (index + totalBanners) % totalBanners;
      for (let i = 0; i < totalBanners; i++) {
        const slide = document.getElementById(`bannerSlide${i}`);
        const dot = document.getElementById(`bannerDot${i}`);
        if (i === currentBannerIndex) {
          slide.classList.add('active');
          dot.className = 'w-6 h-2 rounded-full bg-stpOrange transition-all';
        } else {
          slide.classList.remove('active');
          dot.className = 'w-2.5 h-2 rounded-full bg-white/60 transition-all';
        }
      }
    }

    function changeBannerSlide(step) {
      showBannerSlide(currentBannerIndex + step);
      resetBannerTimer();
    }

    function goToBannerSlide(index) {
      showBannerSlide(index);
      resetBannerTimer();
    }

    function resetBannerTimer() {
      if (bannerInterval) clearInterval(bannerInterval);
      bannerInterval = setInterval(() => {
        showBannerSlide(currentBannerIndex + 1);
      }, 5000);
    }
